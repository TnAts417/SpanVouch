# SpanVouch v0.7.1 Provider-Test Isolation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `executing-plans` to implement this plan task-by-task. Use `test-driven-development`, `systematic-debugging` for unexpected failures, `pdf` for the report artifact, and `verification-before-completion` before any completion claim.

**Goal:** Make pytest and CI fail closed against inherited provider credentials and network access, then prepare a truthful v0.7.1 candidate with a freshly verified engineering snapshot.

**Architecture:** Test policy lives entirely in the root pytest configuration. Provider credentials are removed before test-module imports, an autouse fixture blocks DNS/socket I/O, and CI proves the boundary with non-secret sentinel values. Production composition remains unchanged; release prose is refreshed only from structured outputs of the final offline verification run.

**Tech Stack:** Python 3.12, pytest 8, httpx MockTransport, GitHub Actions, uv 0.8.15, Ruff, mypy, LaTeX/BibTeX.

## Global Constraints

- Target version is `0.7.1`; prepare a local candidate only.
- Never print, persist, or inspect a provider credential value.
- Do not call DeepSeek, Qwen, another provider, or any external network service.
- Do not change production provider composition or live-authorization semantics.
- CI may use only the literal value `offline-test-sentinel-not-a-credential` for provider credential variables and must not reference GitHub secrets.
- Keep B0-B3 DeepSeek-only, B4/B5 `policy-skipped`, no-Qwen status, and H1-H5 `unresolved` unchanged.
- Derive validation counts and coverage from the final run; never guess them.
- Do not push, merge, tag, or create a GitHub Release without explicit publication approval.

```python
from __future__ import annotations

import os
import socket
from collections.abc import Callable

import httpx
import pytest

PROVIDER_CREDENTIAL_ENV_NAMES = (
    "DEEPSEEK_API_KEY",
    "SPANVOUCH_QWEN_API_KEY",
    "OPENAI_API_KEY",
    "QWEN_API_KEY",
)
VISIBLE_DURING_COLLECTION = {
    name: os.environ.get(name) for name in PROVIDER_CREDENTIAL_ENV_NAMES
}


def _socket_connect() -> object:
    with socket.socket() as client:
        return client.connect(None)  # type: ignore[arg-type]


def _socket_connect_ex() -> object:
    with socket.socket() as client:
        return client.connect_ex(None)  # type: ignore[arg-type]


@pytest.mark.parametrize(
    "operation",
    (
        lambda: socket.getaddrinfo("localhost", 80),
        lambda: socket.create_connection(None),  # type: ignore[arg-type]
        _socket_connect,
        _socket_connect_ex,
    ),
)
def test_offline_suite_blocks_every_network_entry_point(
    operation: Callable[[], object],
) -> None:
    with pytest.raises(AssertionError, match="offline test attempted network access"):
        operation()


def test_provider_credentials_are_absent_during_test_collection() -> None:
    assert VISIBLE_DURING_COLLECTION == {
        name: None for name in PROVIDER_CREDENTIAL_ENV_NAMES
    }


@pytest.mark.asyncio
async def test_mock_transport_remains_available_without_socket_access() -> None:
    transport = httpx.MockTransport(
        lambda request: httpx.Response(200, json={"url": str(request.url)})
    )
    async with httpx.AsyncClient(transport=transport) as client:
        response = await client.get("https://provider.invalid/health")

    assert response.json() == {"url": "https://provider.invalid/health"}
```

```python
from __future__ import annotations

import os
import socket
from typing import NoReturn

import pytest

_PROVIDER_CREDENTIAL_ENV_NAMES = (
    "DEEPSEEK_API_KEY",
    "SPANVOUCH_QWEN_API_KEY",
    "OPENAI_API_KEY",
    "QWEN_API_KEY",
)

for _credential_name in _PROVIDER_CREDENTIAL_ENV_NAMES:
    os.environ.pop(_credential_name, None)


def _forbid_network(*args: object, **kwargs: object) -> NoReturn:
    raise AssertionError("offline test attempted network access")


@pytest.fixture(autouse=True)
def _offline_network_guard(monkeypatch: pytest.MonkeyPatch) -> None:
    monkeypatch.setattr(socket, "getaddrinfo", _forbid_network)
    monkeypatch.setattr(socket, "create_connection", _forbid_network)
    monkeypatch.setattr(socket.socket, "connect", _forbid_network)
    monkeypatch.setattr(socket.socket, "connect_ex", _forbid_network)
```
